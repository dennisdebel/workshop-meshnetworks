    // -----------------------
    // FILE START

    if(msg.startsWith("FILE_START:"))
    {
        Serial.print("proxyTransfer=");

        int p1 = msg.indexOf(':', 11);
        int p2 = msg.indexOf(':', p1 + 1);

        String filename = msg.substring(11, p1);
        int filesize = msg.substring(p1 + 1, p2).toInt();
        String mime = msg.substring(p2 + 1);

        Serial.println("FILE:");
        Serial.println(filename);
        Serial.println(filesize);
        Serial.println(mime);

        // -----------------------
        // PROXY TRANSFER
        // -----------------------
        if(proxyTransfer)
        {
            Serial.println("Proxy receiving file via chunked stream");

            proxyMime = mime;
            proxyExpected = filesize;
            proxyLength = 0;

            if(proxyMime.startsWith("image") || proxyMime.startsWith("application"))
            {
                // Ensure any old buffer is cleared
                if(proxyBuffer) {
                    free(proxyBuffer);
                    proxyBuffer = nullptr;
                }
                
                // Allocate a small 1KB sliding queue buffer instead of buffering the whole image!
                proxyBuffer = (uint8_t*)malloc(1024);
            }
            else
            {
                // html/css/js
                proxyData = "";
            }

            return;
        }

        // -----------------------
        // NORMAL FILE RECEIVE
        // -----------------------
        incoming.name = filename;
        incoming.size = filesize;
        incoming.received = 0;

        receivingFile = true;

        String tempName;
        tempName.reserve(80);
        tempName = incoming.name;

        if(!tempName.endsWith(".tmp")) {
            tempName += ".tmp";
        }

        if(getFreeSpace() < incoming.size) {
            Serial.println("Not enough storage!");
            return;
        }

        incoming.file = LittleFS.open(tempName, "w");

        Serial.println("Receiving:");
        Serial.println(incoming.name);

        return;
    }

One more important thing:

Your new design is basically turning the ESP into a reverse proxy. That means the remote node's index.html should contain links like:

<link rel="stylesheet" href="/mesh/NODE/style.css">
<script src="/mesh/NODE/app.js"></script>

The cleanest solution is actually to rewrite those paths in C++ before sending.

The browser should never know it is a mesh. It should just see normal URLs.
